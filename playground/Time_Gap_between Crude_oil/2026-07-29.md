create temporary table join_20260729 as 
-- step1 "휘발유" 가격 가져오기. 이 유종별 전국 평균 가격은 2008-04-15 부터 있다.
with step1 as (
select
strptime (cast(t1.part_dt as varchar) , '%Y%m%d') as part_dt
, t1.prodcd
, t1.prodnm
, t1.price
from read_parquet('s3://petroleum-project/national_avg/price/*/*.parquet') t1
where 1=1  
and t1.part_dt >= '20080415'
and t1.part_dt <= '20260729'
and t1.PRODCD = 'B027' -- 일단은 휘발유 단일 가격만.
)
, step2 as (
select * 
from read_parquet('s3://petroleum-project/crude_oil_price/*/*.parquet') t1
where 1=1
and t1.part_dt >= '2008-04-15'
and t1.part_dt <= '2026-07-29'
and t1.currency = 'USD'
)
select
t1.part_dt
, t1.prodcd
, t1.prodnm
, t1.price
, t2.currency
, t2.unit
, t2.Dubai
, t2.Brent
, t2.WTI
from step1 t1
left join step2 t2
on t1.part_dt= t2.part_dt

 


;
select 
corr ( price, Dubai) as corr_dubai
, corr ( price, Brent) as corr_brent
, corr ( price, WTI) as corr_wti
, count( distinct case when Dubai IS NULL then part_dt end ) as dubai_null_days
, count( distinct case when Brent IS NULL then part_dt end ) as Brent_null_days
, count( distinct case when WTI IS NULL then part_dt end ) as WTI_null_days
from join_20260729

;
